#include <iostream>
#include <thread>
#include <vector>
#include <atomic>
#include <new>

// 定义常量：循环次数，确保运行时间足够 perf 采样
const size_t ITERATIONS = 500000000;

// 1. 伪共享结构体：两个变量挤在一个 Cache Line (64字节)
struct FalseSharing {
    volatile long a; // 线程 1 修改
    volatile long b; // 线程 2 修改
};

// 2. 真共享结构体：两个线程修改同一个变量
struct TrueSharing {
    volatile long a;
};

// 3. 无共享结构体：通过对齐避开同一个 Cache Line
struct NoSharing {
    alignas(64) volatile long a;
    alignas(64) volatile long b;
};

// 全局实例
FalseSharing fs_data;
TrueSharing ts_data;
NoSharing ns_data;

void work_false_a() { for (size_t i = 0; i < ITERATIONS; ++i) fs_data.a++; }
void work_false_b() { for (size_t i = 0; i < ITERATIONS; ++i) fs_data.b++; }

void work_true_a() { for (size_t i = 0; i < ITERATIONS; ++i) ts_data.a++; }

void work_no_a() { for (size_t i = 0; i < ITERATIONS; ++i) ns_data.a++; }
void work_no_b() { for (size_t i = 0; i < ITERATIONS; ++i) ns_data.b++; }

int main(int argc, char* argv[]) {
    if (argc < 2) {
        std::cout << "Usage: " << argv[0] << " <mode: 1-False, 2-True, 3-No>" << std::endl;
        return 1;
    }

    int mode = std::stoi(argv[1]);
    std::vector<std::thread> threads;

    if (mode == 1) {
        std::cout << "Running False Sharing mode..." << std::endl;
        threads.emplace_back(work_false_a);
        threads.emplace_back(work_false_b);
    } else if (mode == 2) {
        std::cout << "Running True Sharing mode..." << std::endl;
        threads.emplace_back(work_true_a);
        threads.emplace_back(work_true_a); // 两个线程竞争同一个 a
    } else {
        std::cout << "Running No Sharing (Aligned) mode..." << std::endl;
        threads.emplace_back(work_no_a);
        threads.emplace_back(work_no_b);
    }

    for (auto& t : threads) t.join();
    std::cout << "Done." << std::endl;

    return 0;
}

use std::fs::File;

#[derive(Debug, Clone)]
struct Event {
    cpu: u32,
    level: String,
    addr: u64, 
    raw_line: String,
}

fn main() {
    let file_path = "/home/neu/exp/perf.txt";
    let mut events: Vec<Event> = Vec::new();

    println!("📥 开始读取并解析日志数据...");

    if let Ok(lines) = read_lines(file_path) {
        for line_result in lines {
            let line = line_result.expect("读取行失败");
            
            if !line.contains("sharing_bench") { continue; }

            if let Some(event) = parse_line(&line) {
                if event.addr > 0x10000 { 
                    events.push(event);
                }
            }
        }
    } else {
        eprintln!("无法读取文件 {}", file_path);
        return;
    }

    println!("🔍 成功加载 {} 条精确访问记录，开始统计跨核 Ping-Pong 效应...\n", events.len());

    // 【核心升级】：使用 HashMap 进行全局统计
    // Key 是元组 (较小地址, 较大地址)，确保 (a,b) 和 (b,a) 都会落入同一个 bucket
    // Value 是该地址组合发生跨核争用的次数
    let mut pair_counts: HashMap<(u64, u64), usize> = HashMap::new();

    for i in 0..events.len() {
        let current_event = &events[i];
            
        for j in (0..i).rev() {
            let prev_event = &events[j];
            
            if prev_event.cpu != current_event.cpu {
                let distance = current_event.addr.abs_diff(prev_event.addr);
                
                if distance < 64 && distance > 0 {
                    // 【规范化地址对】：保证小的地址在前面
                    let min_addr = std::cmp::min(current_event.addr, prev_event.addr);
                    let max_addr = std::cmp::max(current_event.addr, prev_event.addr);
                    
                    // 对应的统计次数 +1
                    *pair_counts.entry((min_addr, max_addr)).or_insert(0) += 1;
                    
                    break; // 找到最近的诱因后，停止回溯
                }
            }
        }
    }

    // 【美化输出】：将结果转为 Vec，并按争用次数从大到小排序
    let mut sorted_results: Vec<_> = pair_counts.into_iter().collect();
    sorted_results.sort_by(|a, b| b.1.cmp(&a.1));

    println!("==================================================");
    println!("🚨 伪共享 (Cacheline 争用) 最终统计报告");
    println!("==================================================");
    
    if sorted_results.is_empty() {
        println!("✅ 未检测到满足条件的伪共享地址对。");
    } else {
        for ((addr1, addr2), count) in sorted_results {
            let distance = addr2 - addr1; // 因为 addr2 肯定大于 addr1，直接相减即可
            println!(
                "💥 [地址对]: 0x{:x} 与 0x{:x} | 间距: {:02} 字节 | 跨核争用: {} 次",
                addr1, addr2, distance, count
            );
        }
    }
    println!("==================================================");
}

fn parse_line(line: &str) -> Option<Event> {
    if !line.contains("|OP STORE|") && !line.contains("|OP LOAD|") { return None; }

    let cpu_start = line.find('[')? + 1;
    let cpu_end = line.find(']')?;
    let cpu: u32 = line[cpu_start..cpu_end].parse().ok()?;

    let lvl_start = line.find("|LVL ")? + 5;
    let lvl_end = line[lvl_start..].find('|')? + lvl_start;
    let level = line[lvl_start..lvl_end].trim().to_string();

    let after_p = line.split("P:").nth(1)?;
    let before_pipe = after_p.split('|').next()?;
    let addr_str = before_pipe.split_whitespace().next()?; 
    
    let addr = u64::from_str_radix(addr_str, 16).unwrap_or(0);

    Some(Event { cpu, level, addr, raw_line: line.to_string() })
}

fn read_lines<P>(filename: P) -> io::Result<io::Lines<io::BufReader<File>>> where P: AsRef<Path> {
    let file = File::open(filename)?;
    Ok(io::BufReader::new(file).lines())
}

## perf.txt 格式

`N/A` 和 `None` 的区别
- **`N/A` (Not Applicable / Not Available)**：**硬件没有收集、或者无法收集这个维度的数据。**
    - 比如在 `mem-stores` (写入) 事件中，你看到 `SNP N/A`。因为对于某些 Intel 架构，写入操作的精确 PEBS 采样不支持记录完整的嗅探 (Snoop) 状态，或者 `perf` 解析时认为该字段不适用，所以直接填了 `N/A`。
- **`None`**：**硬件进行了监测，并且明确记录下“没有发生该行为”。**
    - 比如 `SNP None`。这意味着 CPU 确实向总线发出了内存请求，也确实去其他核心的缓存“嗅探” (Snoop) 了一圈，但**结果是没有在其他核心找到有效的数据包**。这表示该缓存行可能只在主存里，或者在更底层的共享 L3 缓存里。

### 1. 第一列：`[006]` (CPU 核心编号)

- **含义**：这条指令是在哪个逻辑 CPU 核心上运行的。
    
- **在 False Sharing 中的作用**：如果你看到 CPU `[006]` 和 CPU `[005]` 频繁出现在同一个地址段，说明这两个物理核心在“打架”。
    

### 2. 第二列：`cpu/mem-loads...` (事件类型)

- **含义**：记录的采样事件名称。
    
    - `mem-loads`：内存读取采样。
        
    - `mem-stores`：内存写入采样。
        
    - `ldlat=30`：表示只采样延迟超过 30 个周期的加载操作。
        
- **注意**：写入操作（Stores）对伪共享的杀伤力比读取大得多，因为写入会强制使其他核心的缓存行失效。
    

### 3. 第三列：`586ef3671210` (数据逻辑地址)

- **含义**：**这是最重要的信息。** 指令访问的具体变量在内存中的**虚拟地址 (VA)**。
    
- **分析**：你会发现 `work_false_a` 访问的是 `...1210`，而 `work_false_b` 访问的是 `...1218`。它们极其接近，这直接证明了它们共享同一个缓存行（Cache Line 段通常以 64 字节对齐）。
    

### 4. 第四列：`11868100242` (原始元数据)

- **含义**：这是 CPU 硬件传回来的原始 16 进制编码，包含了该次内存访问的所有底层细节。
    
- **分析**：普通用户不需要看这一列，因为它就是下一列“解码内容”的原始版本。
    

### 5. 第五列：`|OP ... |SNP ... |` (解码后的内存属性)

这是 `perf c2c` 提供的最精华的解码信息，包含几个子字段：

- **OP (Operation)**：`LOAD` (读) 或 `STORE` (写)。
    
- **LVL (Level)**：数据从哪级缓存拿到的。
    
    - `L1 hit`：L1 缓存命中。
        
    - `LFB/MAB hit`：行填充缓冲区命中，说明 L1 没中且数据正在传输。
        
- **SNP (Snoop/嗅探)**：**检测共享的关键！**
    
    - `SNP None`：没有发生跨核嗅探。
        
    - `SNP HITM`：**（重点）** 命中了另一个核心修改过的缓存。这就是伪共享的最直接证据。
        
- **TLB**：页表缓冲命中情况。
    
- **LCK (Lock)**：是否带 `LOCK` 前缀的原子操作（如 `atomic++`）。
    

### 6. 第六列：`586ef366e830` (指令逻辑地址)

- **含义**：触发这次内存访问的那行**汇编指令**在内存中的位置（RIP 寄存器的值）。
    
- **作用**：当你拿这个地址去查汇编代码时，能精确找到是哪一条指令（比如 `mov` 或 `add`）触发了缓存竞争。
    

### 7. 第七、八列：`work_false_a() (DSO)` (符号与文件名)

- **含义**：该指令所属的函数名和二进制文件路径。
    
- **作用**：帮你把枯燥的地址映射回人类能看懂的代码逻辑。

**你的理解完全正确！你精准地总结了 False Sharing 的受害者模型。**

这里记录的 `L1 miss` 或 `LFB hit`，确实是**受害者 (Victim)** 在承担延迟，而不是肇事者 (Culprit) 在搞破坏。
- **肇事过程 (无延迟)**：当 Core 006 (运行 `work_false_a`) 执行写入时，它向总线广播一个 Invalid (失效) 信号。这一步对 Core 006 来说通常是非常快的，它把值丢进 Store Buffer 就继续往下跑了，它**不会**经历严重的 Miss（除非它自己原本连缓存都没有）。
- **受害过程 (被抓拍的 MISS)**：此时，Core 005 (运行 `work_false_b`) 对应的 L1 缓存行已经被 Core 006 变成了 **Invalid (无效) 状态**。当 Core 005 下一次想要执行 `fs_data.b++`，它去自己的 L1 缓存一摸，发现数据是 Invalid 的！
- **结果**：Core 005 被迫停下来（Stall），去 L2/L3 甚至 Core 006 的 L1 里把数据重新拉回来。**这个漫长的等待过程，被 CPU 的性能监控单元 (PMU) 抓拍到了**，并在 `perf.txt` 中记录为一个耗时很长（比如 `ldlat=30`，延迟大于 30 个周期）的 `mem-loads` L1 MISS 事件。

这是一个非常硬核且直击灵魂的问题！在性能优化的实战中，找到“受害者”（LOAD MISS）只是第一步，抓到“肇事者”（引起失效的 STORE）才是破案的关键。

首先要告诉你一个残酷的硬件现实：
在 CPU 的底层硬件中，没有任何机制会记录“当前这个 LOAD MISS 是由过去哪条具体的 STORE 指令造成的”。缓存一致性协议（如 MESI）只维护状态（比如这个 Cache Line 是 Invalid 还是 Modified），但它不会在 Cache Line 上贴一个标签说：“是 Core 0 上的指令 IP 0x400abc 把我改掉的”。

但是，不要灰心！虽然硬件没有直接的 1对1 因果映射，我们完全可以通过 perf c2c 的统计学关联和地址对齐机制，精准地把“肇事者”揪出来。

以下是三种最有效的追踪方法，按方便程度排序：

方法一：利用 perf c2c report 的“同伙聚合”功能（最推荐）
perf c2c 工具的伟大之处，就在于它在后台自动帮你做了关联。它把所有访问同一个物理 Cache Line (64字节) 的读写操作，全部聚拢在了一张表里。

操作步骤：

运行 sudo perf c2c report，进入交互界面。

找到 Rmt HitM 最高的那个 Cache Line，按 d 展开详情。

寻找“肇事者”的逻辑：

看受害者： 找到带有 HITM 标记的行，注意它的 Offset（比如 0x10）和 OP（通常是 LOAD）。

找肇事者： 在同一个页面的列表中，往下扫，专门看 OP 列为 STORE 的行！

比对 Offset： 如果你发现有一个 STORE 操作，它的 Offset 是 0x18（和受害者的 0x10 挨得很近，且在同一个表里），这个 STORE 就是导致那个 LOAD MISS 的罪魁祸首！

定位代码： 直接看这个 STORE 对应的 Code address 和 Symbol（函数名），配合 addr2line，你就能精确找到是哪行代码在写入。

方法二：在 perf script 中通过“时间窗口 + 地址匹配”推理
如果你习惯看 perf script 的原始日志，你可以像福尔摩斯一样进行时间线推理。

推理逻辑：

锁定案发地点： 你通过 HITM 记录，发现受害变量在地址 0x...1210。

计算 Cache Line 范围： 现代 CPU 的 Cache Line 是 64 字节对齐的。地址 0x...1210 所在的 Cache Line 范围是 0x...1200 到 0x...123F。

向前追溯（关键步）： 在 perf script 日志中，从那个 HITM 的时间点往上翻。

抓捕嫌疑人： 使用 grep 搜索这 64 字节范围内的地址（例如 grep "1218" perf.txt）。如果你发现在这个 HITM 发生之前不久，有其他 CPU 核心（比如 [005]）对 0x...1218 发生了一个 mem-stores（写入）事件，那么这个写入就是肇事者。

(注：如前所述，Intel 硬件对 STORE 地址的采样可能不准，所以方法一中 perf c2c 基于 Cache Line 基地址的聚合分析更靠谱。)

方法三：终极杀器 —— 硬件数据断点 (Data Watchpoint / GDB)

操作步骤：

用 GDB 挂载你的程序：gdb ./your_program

在受害者变量的内存地址附近打上硬件写断点 (Hardware Watchpoint)。假设你怀疑 fs_data.b 是肇事者修改的变量：

Code snippet

运行程序 (gdb) c

在 x86 架构中，像 `fs_data.a++` 这样的自增操作，在汇编层面是“读-改-写”。当 CPU 006 准备执行这个操作时，它必须先拿到这个缓存行的**独占权 (Exclusive/Modified)**。即使它表面上执行的是一个 `LOAD`（把数据读进寄存器准备加 1），硬件底层其实发出了一个叫 **RFO (Read For Ownership)** 的请求。这个 RFO 请求会直接把其他核心里的缓存行给“干掉”（Invalidate）。

在 Linux C/C++ 程序的内存布局中，`_end` 符号（Address: `0x5e66d8`）确实是一个极其重要的分水岭。它标志着 **BSS 段（未初始化数据段）的结束**，也就是说，程序中所有静态分配的全局变量（包括你的 `fs_data`、`ts_data`）的最大地址绝对不会超过这个值。

超过 `_end` 的地址只可能是：

1. **Heap (堆)**：通过 `malloc`/`new` 动态分配的内存。
2. **Stack (栈)**：局部变量（通常在极高的地址，比如 `0x7fff...`）。
3. **mmap 映射区**：动态链接库或大块内存。
4. **硬件采样的杂音 / 越界地址**。
